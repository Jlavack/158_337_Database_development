create table branch (br_branchno varchar(4), 
br_branchname varchar(20),
br_street varchar(40) not null unique,
br_city varchar(20), br_state varchar(2), br_zipcode varchar(5),
br_phone varchar(10), 
br_fax varchar(10),
br_per_mgt_fee number,
br_email varchar(25),
constraint br_primarykey primary key (br_branchno));

create table staff (st_staffno varchar(4) primary key,
st_name varchar(40), 
st_position varchar(20), 
st_sex char(1), 
st_birthdate date,
st_hiredate date,
st_salary number(6),
st_commperc number, 
st_br_branchno varchar(4),
constraint fk_staff1 foreign key (st_br_branchno) references branch (br_branchno));
